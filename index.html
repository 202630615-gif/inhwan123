        class SoundEngine {
            constructor() {
                this.ctx = null;
                this.sfxMuted = false;
                this.bgmMuted = false;
                this.bgmInterval = null;
                this.bgmStep = 0;
            }

            init() {
                if (!this.ctx) {
                    const AudioContext = window.AudioContext || window.webkitAudioContext;
                    this.ctx = new AudioContext();
                }
                if (this.ctx.state === 'suspended') {
                    this.ctx.resume();
                }
            }

            playTone(freq, type = 'sine', duration = 0.1, vol = 0.1) {
                if (this.sfxMuted || !this.ctx) return;
                try {
                    const osc = this.ctx.createOscillator();
                    const gain = this.ctx.createGain();
                    osc.type = type;
                    osc.frequency.setValueAtTime(freq, this.ctx.currentTime);
                    gain.gain.setValueAtTime(vol, this.ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.0001, this.ctx.currentTime + duration);
                    osc.connect(gain);
                    gain.connect(this.ctx.destination);
                    osc.start();
                    osc.stop(this.ctx.currentTime + duration);
                } catch(e) {}
            }

            playShoot() {
                if (this.sfxMuted || !this.ctx) return;
                try {
                    const osc = this.ctx.createOscillator();
                    const gain = this.ctx.createGain();
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(700, this.ctx.currentTime);
                    osc.frequency.exponentialRampToValueAtTime(100, this.ctx.currentTime + 0.12);
                    gain.gain.setValueAtTime(0.12, this.ctx.currentTime);
                    gain.gain.linearRampToValueAtTime(0.01, this.ctx.currentTime + 0.12);
                    osc.connect(gain);
                    gain.connect(this.ctx.destination);
                    osc.start();
                    osc.stop(this.ctx.currentTime + 0.12);
                } catch(e) {}
            }

            playExplosion() {
                if (this.sfxMuted || !this.ctx) return;
                try {
                    const bufferSize = this.ctx.sampleRate * 0.3;
                    const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
                    const data = buffer.getChannelData(0);
                    for (let i = 0; i < bufferSize; i++) {
                        data[i] = Math.random() * 2 - 1;
                    }
                    const noise = this.ctx.createBufferSource();
                    noise.buffer = buffer;

                    const filter = this.ctx.createBiquadFilter();
                    filter.type = 'lowpass';
                    filter.frequency.setValueAtTime(900, this.ctx.currentTime);
                    filter.frequency.linearRampToValueAtTime(40, this.ctx.currentTime + 0.3);

                    const gain = this.ctx.createGain();
                    gain.gain.setValueAtTime(0.25, this.ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.3);

                    noise.connect(filter);
                    filter.connect(gain);
                    gain.connect(this.ctx.destination);
                    noise.start();
                } catch(e) {}
            }

                    p.alpha -= p.decay;

                        this.particles.splice(i, 1);
                    }
                }

                    ctx.shadowBlur = 10;
                    ctx.shadowColor = p.color;
                    ctx.beginPath();
                    ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
                    ctx.fill();
                }
                ctx.restore();

                this.particles = [];

                            this.bullets.splice(j, 1);
                            globalParticles.spawn(b.x, b.y, '#ffff00', 6);
                            if (e.hp <= 0) break;
                        }
                    }

                    // Enemy Destroyed
                    if (e.hp <= 0) {
                        sounds.playExplosion();
                        screenShake = 6;
                        globalParticles.spawn(e.x, e.y, e.color, 16, 1.4);
                        score += e.type === 'heavy' ? 250 : 100;

                        if (Math.random() < 0.18) {
                            const pTypes = ['TRIPLE', 'SHIELD', 'HEAL'];
                            this.powerups.push({
                                x: e.x, y: e.y,
                                type: pTypes[Math.floor(Math.random() * pTypes.length)],
                                vy: 2
                            });
                        }
                        this.enemies.splice(i, 1);
                        continue;
                    }

                    // Collision with Player
                    if (Math.hypot(e.x - p.x, e.y - p.y) < (e.width + p.width) / 2.8) {
                        if (p.shieldTimer <= 0) {
                            p.hp--;
                            sounds.playExplosion();
                            screenShake = 15;
                            globalParticles.spawn(p.x, p.y, '#ff0000', 30);
                            if (p.hp <= 0) endGame();
                        } else {
                            sounds.playBounce();
                        }
                        this.enemies.splice(i, 1);
                        continue;
                    }

                    if (e.y > GAME_HEIGHT + 40) this.enemies.splice(i, 1);
                }

                // Powerups Update
                for (let i = this.powerups.length - 1; i >= 0; i--) {
                    const pow = this.powerups[i];
                    pow.y += pow.vy;
                    if (Math.hypot(pow.x - p.x, pow.y - p.y) < 32) {
                        sounds.playPowerup();
                        if (pow.type === 'TRIPLE') p.tripleShotTimer = 450;
                        if (pow.type === 'SHIELD') p.shieldTimer = 400;
                        if (pow.type === 'HEAL') p.hp = Math.min(p.maxHp, p.hp + 1);
                        globalParticles.spawn(pow.x, pow.y, '#00ff88', 15);
                        this.powerups.splice(i, 1);
                    } else if (pow.y > GAME_HEIGHT) {
                        this.powerups.splice(i, 1);
                    }
                }
            }

            fireBullet() {
                sounds.playShoot();
                const p = this.player;
                if (p.tripleShotTimer > 0) {
                    this.bullets.push({ x: p.x, y: p.y - 22, vy: -13, vx: 0, radius: 4 });
                    this.bullets.push({ x: p.x - 14, y: p.y - 18, vy: -12, vx: -2.2, radius: 4 });
                    this.bullets.push({ x: p.x + 14, y: p.y - 18, vy: -12, vx: 2.2, radius: 4 });

                }
            }

        class BrickBreakerGame {
            constructor() {
                this.paddle = {
                    width: 130,
                    height: 18,
                    x: GAME_WIDTH / 2 - 65,
                    y: GAME_HEIGHT - 45,
                    speed: 9.5
                };
                this.balls = [{
                    x: GAME_WIDTH / 2,
                    y: GAME_HEIGHT - 65,
                    vx: 4.5 * (Math.random() > 0.5 ? 1 : -1),
                    vy: -5.5,
                    radius: 8
                }];
                this.bricks = [];
                this.powerups = [];
                this.initBricks();
            }

            initBricks() {
                const rows = 5;
                const cols = 9;
                const brickWidth = 76;
                const brickHeight = 22;
                const padding = 8;
                const offsetTop = 65;
                const offsetLeft = (GAME_WIDTH - (cols * (brickWidth + padding))) / 2;

                const colors = ['#ff007f', '#00f0ff', '#ffff00', '#00ff88', '#aa00ff'];

                for (let r = 0; r < rows; r++) {
                    for (let c = 0; c < cols; c++) {
                        this.bricks.push({
                            x: offsetLeft + c * (brickWidth + padding),
                            y: offsetTop + r * (brickHeight + padding),
                            width: brickWidth,
                            height: brickHeight,
                            color: colors[r % colors.length],
                            hp: r === 0 ? 2 : 1
                        });
                    }
                }
            }

            update() {
                const p = this.paddle;
                if ((keys['ArrowLeft'] || keys['KeyA'] || touchState.left) && p.x > 0) p.x -= p.speed;
                if ((keys['ArrowRight'] || keys['KeyD'] || touchState.right) && p.x < GAME_WIDTH - p.width) p.x += p.speed;

                if (isMouseDown && mouseX > 0 && mouseX < GAME_WIDTH) {
                    p.x = mouseX - p.width / 2;
                }
                if (p.x < 0) p.x = 0;
                if (p.x > GAME_WIDTH - p.width) p.x = GAME_WIDTH - p.width;

                for (let i = this.balls.length - 1; i >= 0; i--) {
                    const b = this.balls[i];
                    b.x += b.vx;
                    b.y += b.vy;

                    // Wall Collision
                    if (b.x - b.radius < 0 || b.x + b.radius > GAME_WIDTH) {
                        b.vx *= -1;
                        sounds.playBounce();
                    }
                    if (b.y - b.radius < 0) {
                        b.vy *= -1;
                        sounds.playBounce();
                    }

                    // Paddle Collision
                    if (b.y + b.radius >= p.y && b.y - b.radius <= p.y + p.height &&
                        b.x >= p.x && b.x <= p.x + p.width) {
                        
                        const hitPoint = (b.x - (p.x + p.width / 2)) / (p.width / 2);
                        const speed = Math.hypot(b.vx, b.vy);
                        b.vx = hitPoint * (speed * 0.85);
                        b.vy = -Math.sqrt(Math.max(12, speed * speed - b.vx * b.vx));
                        sounds.playBounce();
                    }

                    // Bricks Collision
                    for (let j = this.bricks.length - 1; j >= 0; j--) {
                        const br = this.bricks[j];
                        if (b.x + b.radius > br.x && b.x - b.radius < br.x + br.width &&
                            b.y + b.radius > br.y && b.y - b.radius < br.y + br.height) {
                            
                            br.hp--;
                            b.vy *= -1;
                            sounds.playScore();
                            globalParticles.spawn(b.x, b.y, br.color, 10);

                            if (br.hp <= 0) {
                                score += 150;
                                if (Math.random() < 0.22) {
                                    this.powerups.push({
                                        x: br.x + br.width / 2,
                                        y: br.y,
                                        type: Math.random() < 0.5 ? 'EXPAND' : 'MULTIBALL',
                                        vy: 2.2
                                    });
                                }
                                this.bricks.splice(j, 1);
                            }
                            break;
                        }
                    }

                    if (b.y > GAME_HEIGHT + 20) {
                        this.balls.splice(i, 1);
                    }
                }

                if (this.balls.length === 0) endGame();
                if (this.bricks.length === 0) {
                    sounds.playPowerup();
                    this.initBricks();
                }

                // Powerup Collection
                for (let i = this.powerups.length - 1; i >= 0; i--) {
                    const pow = this.powerups[i];
                    pow.y += pow.vy;
                    if (pow.y >= p.y && pow.y <= p.y + p.height && pow.x >= p.x && pow.x <= p.x + p.width) {
                        sounds.playPowerup();
                        if (pow.type === 'EXPAND') p.width = Math.min(230, p.width + 35);
                        if (pow.type === 'MULTIBALL' && this.balls.length > 0) {
                            const main = this.balls[0];
                            this.balls.push({ x: main.x, y: main.y, vx: -main.vx, vy: main.vy, radius: 8 });
                        }
                        this.powerups.splice(i, 1);
                    } else if (pow.y > GAME_HEIGHT) {
                        this.powerups.splice(i, 1);
                    }
                }
            }

            draw(ctx) {
                this.bricks.forEach(br => {
                    ctx.fillStyle = br.color;
                    ctx.shadowBlur = 10;
                    ctx.shadowColor = br.color;
                    ctx.fillRect(br.x, br.y, br.width, br.height);
                });

                const p = this.paddle;
                ctx.fillStyle = '#00f0ff';
                ctx.shadowBlur = 15;
                ctx.shadowColor = '#00f0ff';
                ctx.fillRect(p.x, p.y, p.width, p.height);

                ctx.fillStyle = '#ffffff';
                ctx.shadowBlur = 12;
                ctx.shadowColor = '#ffffff';
                this.balls.forEach(b => {
                    ctx.beginPath();
                    ctx.arc(b.x, b.y, b.radius, 0, Math.PI * 2);
                    ctx.fill();
                });

                this.powerups.forEach(pow => {
                    ctx.fillStyle = pow.type === 'EXPAND' ? '#00ff88' : '#ff007f';
                    ctx.shadowBlur = 8;
                    ctx.shadowColor = ctx.fillStyle;
                    ctx.font = '12px "Press Start 2P"';
                    ctx.fillText(pow.type[0], pow.x, pow.y);
                });
            }
        }

        class FlappyJetGame {
            constructor() {
                this.bird = {
                    x: 140,
                    y: GAME_HEIGHT / 2,
                    radius: 16,
                    vy: 0,
                    gravity: difficulty === 'EASY' ? 0.38 : 0.48,
                    jump: difficulty === 'EASY' ? -7.8 : -8.8
                };
                this.pipes = [];
                this.spawnTimer = 0;
            }

            update() {
                const b = this.bird;

                if (keys['Space'] || keys['ArrowUp'] || keys['KeyW'] || touchState.action || touchState.up || isMouseDown) {
                    if (!this.jumpHandled) {
                        b.vy = b.jump;
                        sounds.playTone(420, 'sine', 0.08, 0.1);
                        globalParticles.spawn(b.x - 12, b.y, '#ff007f', 5);
                        this.jumpHandled = true;
                    }
                } else {
                    this.jumpHandled = false;
                }

                b.vy += b.gravity;
                b.y += b.vy;

                if (b.y + b.radius > GAME_HEIGHT || b.y - b.radius < 0) {
                    sounds.playExplosion();
                    screenShake = 15;
                    endGame();
                }

                this.spawnTimer++;
                if (this.spawnTimer > 85) {
                    this.spawnTimer = 0;
                    const gap = difficulty === 'EASY' ? 175 : 135;
                    const minH = 60;
                    const topH = Math.random() * (GAME_HEIGHT - gap - minH * 2) + minH;

                    this.pipes.push({
                        x: GAME_WIDTH,
                        topHeight: topH,
                        bottomY: topH + gap,
                        width: 60,
                        passed: false
                    });
                }

                for (let i = this.pipes.length - 1; i >= 0; i--) {
                    const p = this.pipes[i];
                    p.x -= 3.8;

                    if (!p.passed && p.x + p.width < b.x) {
                        p.passed = true;
                        score += 100;
                        sounds.playScore();
                    }

                    if (b.x + b.radius > p.x && b.x - b.radius < p.x + p.width) {
                        if (b.y - b.radius < p.topHeight || b.y + b.radius > p.bottomY) {
                            sounds.playExplosion();
                            screenShake = 15;
                            globalParticles.spawn(b.x, b.y, '#ff0055', 25);
                            endGame();
                        }
                    }

                    if (p.x < -p.width) this.pipes.splice(i, 1);
                }
            }

            draw(ctx) {
                const b = this.bird;
                ctx.save();
                ctx.translate(b.x, b.y);
                ctx.rotate(Math.min(Math.PI / 4, Math.max(-Math.PI / 4, b.vy * 0.08)));

                ctx.fillStyle = '#ffe600';
                ctx.shadowBlur = 15;
                ctx.shadowColor = '#ffe600';
                ctx.beginPath();
                ctx.arc(0, 0, b.radius, 0, Math.PI * 2);
                ctx.fill();

                ctx.fillStyle = '#ff007f';
                ctx.fillRect(-10, -5, 14, 9);
                ctx.restore();

                this.pipes.forEach(p => {
                    ctx.fillStyle = '#00f0ff';
                    ctx.shadowBlur = 12;
                    ctx.shadowColor = '#00f0ff';
                    ctx.fillRect(p.x, 0, p.width, p.topHeight);
                    ctx.fillRect(p.x, p.bottomY, p.width, GAME_HEIGHT - p.bottomY);
                });
            }
        }

        function fallbackCopy(text) {
            const textarea = document.createElement('textarea');
            textarea.value = text;
            textarea.style.position = 'fixed';
            textarea.style.opacity = '0';
            document.body.appendChild(textarea);
            textarea.select();
            try {
                document.execCommand('copy');
                showToast();
            } catch (err) {
                alert('복사에 실패했습니다. 수동으로 선택하여 복사해주세요.');
            }
            document.body.removeChild(textarea);
        }

        function gameLoop() {
            ctx.save();

            // Screen Shake Effect
            if (screenShake > 0) {
                const dx = (Math.random() - 0.5) * screenShake;
                const dy = (Math.random() - 0.5) * screenShake;
                ctx.translate(dx, dy);
                screenShake *= 0.88;
                if (screenShake < 0.5) screenShake = 0;
            }

            ctx.clearRect(0, 0, GAME_WIDTH, GAME_HEIGHT);

            if (gameState === 'PLAYING' && currentGameObj) {
                currentGameObj.update();
